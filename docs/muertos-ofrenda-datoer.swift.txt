    /// The day of the Day of the Dead the ofrenda is finished on (2 November; 1 November is for the children).
    static let ofrendaDay = (month: 11, day: 2)

    /// How far the snowman (or what stands in its place) has come on `day`.
    /// It grows a step a day from the first start. The ofrenda is built toward the Day of the Dead instead: a step a
    /// day from 22 October, finished on 2 November, as families build theirs in the last days of October. Outside
    /// those twelve days it grows from the first start like the snowman (so it is also built when the theme is
    /// chosen at another time of year); after 2 November it stands finished if the app was there on the day.
    func stage(on day: Date) -> Int { Self.stage(on: day, theme: theme, firstRun: cfg.firstRun) }
    static func stage(on day: Date, theme: Theme, firstRun: Date?) -> Int {
        let fromStart = max(1, min(snowmanMaxStage, JuleClock.days(from: firstRun ?? day, to: day) + 1))
        guard theme == .muertos else { return fromStart }
        let year = JuleClock.parts(day).year
        guard let finished = Calendar.current.date(from: DateComponents(year: year, month: ofrendaDay.month, day: ofrendaDay.day)) else { return fromStart }
        let left = JuleClock.days(from: day, to: finished)
        if left >= 0 && left < snowmanMaxStage { return snowmanMaxStage - left }
        // after the day, an altar that stood then stays finished (it must not go back to its count from the first start)
        if left < 0, let firstRun, JuleClock.days(from: firstRun, to: finished) >= 0 { return snowmanMaxStage }
        return fromStart
    }
